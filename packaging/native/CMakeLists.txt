cmake_minimum_required(VERSION 3.24)
project(PathMan.SetupActions LANGUAGES CXX RC)
set(PATHMAN_VERSION_NUMBER "0.0.0" CACHE STRING "Numeric build identity")
set(PATHMAN_VERSION_NAME "0.0.0" CACHE STRING "Product version name")
string(REPLACE "." "," PATHMAN_VERSION_TUPLE "${PATHMAN_VERSION_NUMBER}.0")
set(PATHMAN_BINARY_NAME "PathMan.SetupActions")
set(PATHMAN_FILE_TYPE "VFT_DLL")
configure_file(version.rc.in version.rc @ONLY)
add_library(PathMan.SetupActions SHARED actions.cpp "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_compile_features(PathMan.SetupActions PRIVATE cxx_std_20)
target_compile_definitions(PathMan.SetupActions PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PathMan.SetupActions PRIVATE shell32 ole32 uuid advapi32 user32)
set_target_properties(PathMan.SetupActions PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    target_compile_options(PathMan.SetupActions PRIVATE /W4 /WX /utf-8 /permissive-)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(PathMan.SetupPathTests ../../tests/setup-path.cpp)
    target_compile_features(PathMan.SetupPathTests PRIVATE cxx_std_20)
    target_compile_definitions(PathMan.SetupPathTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(PathMan.SetupPathTests PRIVATE PathMan.SetupActions advapi32)
    set_target_properties(PathMan.SetupPathTests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    if(MSVC)
        target_compile_options(PathMan.SetupPathTests PRIVATE /W4 /WX /utf-8 /permissive-)
    endif()
    add_test(NAME setup-path COMMAND PathMan.SetupPathTests)
    add_executable(PathMan.SetupWorkerTests ../../tests/setup-worker.cpp)
    target_compile_features(PathMan.SetupWorkerTests PRIVATE cxx_std_20)
    target_compile_definitions(PathMan.SetupWorkerTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_target_properties(PathMan.SetupWorkerTests PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    if(MSVC)
        target_compile_options(PathMan.SetupWorkerTests PRIVATE /W4 /WX /utf-8 /permissive-)
    endif()
    add_test(NAME setup-worker COMMAND PathMan.SetupWorkerTests)
endif()
